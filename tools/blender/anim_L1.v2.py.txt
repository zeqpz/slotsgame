# on the rig's NLA track, and its spray drawn frame by frame in Grease Pencil on the same timeline.
# (Redone 2026-11-03 for the new upright art; the old spraying-can version is anim_L1.orig.py.txt.)
#   the rig     tip (the root, low on the base's right side: the can tilts over about it), the can on it (its base:
#               sways, squashes, rattles) and the cap on the can (pressed straight down onto the stem)
#               (tools/blender/cuts/L1.py)
#   the spray   drawn frame by frame, only when the cap is pressed: a cone of paint shooting from the nozzle (it
#               rides the nozzle while the can rumbles, sags a little under its own weight, and once the cap is let
#               go its tail leaves the nozzle and flies into the cloud), a cloud of cartoon paint puffs billowing
#               where it ends (solid, inked, glossy - they swell, boil on twos, then drift and shrink away), specks
#               round the jet, droplets flung off the cloud
#   idle        calm, not spraying: it sways on its base, a glint slides down the glossy body, the ball inside
#               rattles once (a little hop and shiver, the cap a beat behind, rattle marks)
#   land        it thuds onto its base (dust), tips over a little and lets out a quick puff, drops back upright
#   hit         it crouches, tips over, the cap is slammed down: a strong burst - long jet, a big boiling cloud,

#   multi       shaken hard (the ball rattles, pulses of light), then it tips over and lets go one big burst with
#               rings of light and streaks; it rights itself by the end
# Every one-shot ends exactly on the rest pose (upright, cap up, no paint left in the air): the idle follows.

def mats(f, bones):
    sc.frame_set(f)
    out = {}
    for b in bones:
        pb = arm.pose.bones[b]
        out[b] = arm.matrix_world @ pb.matrix @ pb.bone.matrix_local.inverted()
    return out

def wrap(f):
    """Idle frames past the loop's end come round to its start (the loop plays seamlessly)."""
    a, b = CLIPS["idle"]
    return a + (f - a) % IDLE_N if a <= f <= b + 40 else f


def on(bone, f, p):
    """World (x, z) of an icon-px point riding a bone at frame f."""
    q = M[f][bone] @ vec(*p)
    return q.x, q.z


def nozzle(f):
    """Where the nozzle is at frame f and which way it points (the cap's own left, unit vector), world (x, z)."""
    Mc = M[f]["cap"]
    p = Mc @ vec(*NOZZLE)
    d = (Mc.to_3x3() @ Vector((-1, 0, 0))).normalized()
    return (p.x, p.z), (d.x, d.z)


def ease(t):
    t = max(0.0, min(1.0, t))
    return 1 - (1 - t) ** 2.2


def clamp_in(x, z, r, lim=6.15):
    """Keep a round shape of radius r inside the 13-unit frame (a sprite is clipped at its edge)."""
    return max(-lim + r, min(lim - r, x)), max(-lim + r, min(lim - r, z))


def dot(board, f, x, z, r, mat):
    if r > .006:
        board.add(f, lambda dr, x=x, z=z, r=r: stroke(dr, ring(x, z, r, n=12), .004, mat=mat, cyclic=True, fill=True, line=False))


def spray(on_f, off_f, L=2.4, w=.36, cloud=1.0, n_puffs=8, seed=0, grow=3, tail=4, linger=10, sag=.12, specks=8):
    """One spray, the cap held down from on_f to off_f: the jet shoots out of the nozzle over `grow` frames and rides
    it; at off_f its tail leaves the nozzle and runs into the cloud over `tail` frames. The cloud: n_puffs inked puffs
    born one after another where the jet ends, swelling and boiling; once the cap is let go they drift off and shrink
    away (staggered, the last gone `linger` frames after the tail)."""
    rnd = random.Random(seed)
    frozen = nozzle(off_f)

    def centre(f, s, frame):
        (nx, nz), (dx, dz) = frame
        return nx + dx * s, nz + dz * s - sag * s * s

    def axis(frame, s):
        a = centre(0, s, frame); b = centre(0, s + .02, frame)
        tx, tz = b[0] - a[0], b[1] - a[1]
        ln = math.hypot(tx, tz) or 1
        return tx / ln, tz / ln

    # the cloud's puffs, laid out round where the jet ends (u along the jet, v across it), the first ones nearest
    puffs = []
    for i in range(n_puffs):
        if i == 0:
            u, v, r = .1, .05, .62
        else:                           # (v > 0 is below the jet: the cloud hangs low, out of the frame's top)
            u = rnd.uniform(.2, 1.3); v = rnd.uniform(-.55, .95) * (.45 + .6 * u)
            r = rnd.uniform(.44, .74) * (1.08 - .22 * abs(v))
        born = on_f + grow - 1 + int(u * 5 + i * .5) + rnd.randint(0, 1)          # the near ones first: it grows outward
        die = off_f + tail + int(linger * (.45 + .55 * rnd.random()))
        drift = (rnd.uniform(.25, .55), rnd.uniform(.15, .45))          # along the jet, and up, as it drifts off
        puffs.append(dict(u=u * cloud, v=v * cloud, r=r * cloud, born=born, die=die, drift=drift, seed=seed * 101 + i))
    puffs.sort(key=lambda p: -p["u"])                                   # the far ones drawn first
    last = max(p["die"] for p in puffs)

    for f in range(on_f, last + 1):
        frame = nozzle(f) if f <= off_f else frozen
        head = L * ease((f - on_f + 1) / grow)
        tl = 0.0 if f <= off_f else L * ease((f - off_f) / tail)
        end = centre(f, L, frame)
        tx, tz = axis(frame, L)
        nx_, nz_ = -tz, tx
        # ---- the cloud
        for p in puffs:
            if not (p["born"] <= f < p["die"]):
                continue
            age = f - p["born"]
            grow_t = min(1.0, age / 5)
            size = (1.12 * math.sin(grow_t * math.pi / 2) if grow_t < 1 else 1.12 - .12 * min(1.0, (age - 5) / 3))
            size *= 1 + .05 * math.sin(f * .9 + p["seed"])                 # it breathes
            gone = max(0.0, (f - off_f) / max(1, p["die"] - off_f)) if f > off_f else 0.0
            size *= (1 - gone ** 1.15)
            dr_u = p["drift"][0] * gone * 2.2 + .12 * min(1.0, age / 20)
            dr_z = p["drift"][1] * gone * 1.4
            cx = end[0] + tx * (p["u"] + dr_u) + nx_ * p["v"]
            cz = end[1] + tz * (p["u"] + dr_u) + nz_ * p["v"] + dr_z
            r = p["r"] * size
            if r < .035:
                continue
            cx, cz = clamp_in(cx, cz, r * 1.2)
            sd = p["seed"] * 7 + (f // 2)                                  # it boils, on twos
            ink = .045 * min(1.0, r / .4) + .01
            # a round lump of paint: inked, its shade low on the right, lit green up on the left, a wet gloss
            air.add(f, lambda dr, cx=cx, cz=cz, r=r, sd=sd, ink=ink: stroke(dr, puff(cx, cz, r, lobes=4, seed=sd, bump=.06), ink,
                                                                             mat=mp["puff"], cyclic=True, fill=True))
            air.add(f, lambda dr, cx=cx, cz=cz, r=r, sd=sd: stroke(dr, puff(cx - .13 * r, cz + .1 * r, r * .8, lobes=4, seed=sd, bump=.06),
                                                                    .004, mat=mp["body"], cyclic=True, fill=True, line=False))
            if r > .18:
                air.add(f, lambda dr, cx=cx, cz=cz, r=r: stroke(dr, puff(cx - .42 * r, cz + .3 * r, r * .15, lobes=3, seed=1, bump=.05),
                                                                 .004, mat=mp["shine"], cyclic=True, fill=True, line=False))
                air.add(f, lambda dr, cx=cx, cz=cz, r=r: stroke(dr, ring(cx - .2 * r, cz + .45 * r, r * .055, n=10),
                                                                 .004, mat=mp["shine"], cyclic=True, fill=True, line=False))
        # ---- the jet: a cone of paint from the nozzle (its tail leaves the nozzle once the cap is let go)
        s0, s1 = tl, head
        if s1 - s0 > .05:
            n = 14
            ss = [s0 + (s1 - s0) * j / (n - 1) for j in range(n)]
            flick = 1 + .07 * math.sin(f * 2.1)
            left, right = [], []
            for s in ss:
                cx, cz = centre(f, s, frame)
                ax, az = axis(frame, s)
                hw = (.07 + (w - .07) * (s / L) ** .85) * flick * (1 + .04 * math.sin(s * 5 + f * 1.3))
                if tl > 0:                                                 # a let-go jet's tail: rounded off to a point
                    hw *= min(1.0, (s - s0) / .45) ** .55
                left.append((cx - az * hw, cz + ax * hw)); right.append((cx + az * hw, cz - ax * hw))
            shape = left + right[::-1]
            if f < on_f + grow + 4 and f <= off_f:                         # the paint's front as it shoots out: a round, inked head
                hx, hz = centre(f, head, frame)
                hr = (.07 + (w - .07) * (head / L) ** .85) * 1.12
                air.add(f, lambda dr, hx=hx, hz=hz, hr=hr: stroke(dr, ring(hx, hz, hr, n=24), .04, mat=mp["puff"], cyclic=True, fill=True))
                air.add(f, lambda dr, hx=hx, hz=hz, hr=hr: stroke(dr, ring(hx - .1 * hr, hz + .08 * hr, hr * .8, n=24), .004, mat=mp["body"],
                                                                   cyclic=True, fill=True, line=False))
            air.add(f, lambda dr, pts=shape: stroke(dr, pts, .004, mat=mp["body"], cyclic=True, fill=True, line=False))
            for side in (left, right):                                     # ink down its sides, thinning into the cloud
                rad = [.04 * min(1.0, (1 - j / (n - 1)) / .4) ** .8 + .004 for j in range(n)]
                air.add(f, lambda dr, pts=side, rad=rad: stroke(dr, pts, .04, mat=mp["ink"], radii=rad))
            hi = [left[j] for j in range(2, n // 2 + 2)]                   # a streak of gloss along its upper side
            hi = [(x + (cx_ - x) * .35, z + (cz_ - z) * .35) for (x, z), (cx_, cz_) in zip(hi, [centre(f, ss[j], frame) for j in range(2, n // 2 + 2)])]
            air.add(f, lambda dr, pts=hi: stroke(dr, pts, .03, mat=mp["shine"], radii=fxkit.taper(len(pts), .032, .4)))
            # specks of paint misting off its sides (new ones every other frame)
            r2 = random.Random(seed * 997 + f // 2)
            for j in range(specks if f <= off_f else specks // 2):
                s = r2.uniform(max(s0, .3), s1)
                cx, cz = centre(f, s, frame)
                ax, az = axis(frame, s)
                hw = .07 + (w - .07) * (s / L) ** .85
                side = r2.choice((-1, 1))
                off = hw + r2.uniform(.04, .22)
                dot(air, f, cx - az * off * side, cz + ax * off * side, r2.uniform(.022, .048), mp["speck"])
    return end, (tx, tz)


def fling(f0, n, seed, x, z, spread=(0, 360), speed=(3, 6), size=(.04, .09), life=12, grav=6.0):
    """Paint droplets flung out: they fly, fall and shrink away."""
    rnd = random.Random(seed)
    for i in range(n):
        a = math.radians(rnd.uniform(*spread)); sp = rnd.uniform(*speed); r0 = rnd.uniform(*size)
        for k in range(life):
            t = k / FPS; u = k / (life - 1)
            r = r0 * (1 - u ** 1.6)
            xx, zz = clamp_in(x + math.cos(a) * sp * t, z + math.sin(a) * sp * t - grav * t * t, r)
            if r > .006:
                air.add(f0 + k, lambda dr, xx=xx, zz=zz, r=r: stroke(dr, ring(xx, zz, r, n=14), .022, mat=mp["drop"], cyclic=True, fill=True))


def rings_out(board, f0, cx, cz, r0, r1, mats_, life=11, lag=3, width=.12):
    """Rings rolling outward at full strength, thinning to nothing (never fading to a dim, dirty line)."""
    for j, mat in enumerate(mats_):
        for k in range(life):
            t = k / (life - 1)
            r = r0 + (r1 - r0) * (1 - (1 - t) ** 1.6)
            w = width * (1 - t) ** 1.3
            if w > .004:
                board.add(f0 + lag * j + k, lambda dr, r=r, w=w, mat=mat, f=f0 + k: stroke(
                    dr, ring(cx, cz, r, n=96, wobble=.02, seed=f), w, mat=mat, cyclic=True))


def ticks(board, f0, x, z, mats_, a0=20, a1=160, n=5, r0=.25, r1=.85, life=6, w=.035, seed=0):
    rnd = random.Random(seed)
    angs = [math.radians(a0 + (a1 - a0) * (i + .5) / n + rnd.uniform(-7, 7)) for i in range(n)]
    for k in range(life):
        t = k / max(1, life - 1)
        for i, a in enumerate(angs):
            ra = r0 + (r1 - r0) * t ** .7
            rb = ra + .3 * (1 - t) + .04
            board.add(f0 + k, lambda dr, a=a, ra=ra, rb=rb, t=t, i=i: stroke(
                dr, [(x + math.cos(a) * ra, z + math.sin(a) * ra), (x + math.cos(a) * rb, z + math.sin(a) * rb)],
                w * (1 - t) ** .8 + .006, mat=mats_[i % len(mats_)]))
